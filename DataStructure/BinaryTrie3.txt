// xor非対応
// verify https://atcoder.jp/contests/abc287/submissions/56218458
// insert, erase, rightkth_query
// val, bit, count
template<typename T = int, typename B = int, typename C = int, int MAXLEN = 31>
struct BinaryTrie {
public:
    struct Node {
        B key;
        T val;
        C cnt;
        int depth;
        int NextNode[2];
        Node(B _key = 0, int _depth = 0) : key(_key), depth(_depth), cnt(0), NextNode{-1, -1} {}
    };
    using pNode = shared_ptr<Node>;
    vector<pNode> pool;
    BinaryTrie() {
        pool.push_back(make_shared<Node>(-1, 0));
    }

    // method
    void insert(B bit, C cnt = 1) {
        int idx = 0;
        stack<int> st;
        st.push(idx);
        while(pool[idx]->depth != MAXLEN) {
            int depth = pool[idx]->depth;
            B nex = (bit >> (MAXLEN - depth - 1));
            if(pool[idx]->NextNode[nex & 1] == -1) {
                pool[idx]->NextNode[nex & 1] = pool.size();
                pool.push_back(make_shared<Node>(nex, depth + 1));
            }
            st.push(pool[idx]->NextNode[nex & 1]);
            idx = pool[idx]->NextNode[nex & 1];
        }
        pool[st.top()]->key = bit;
        pool[st.top()]->cnt += cnt;
        pool[st.top()]->val = bit * pool[st.top()]->cnt;
        st.pop();
        while(!st.empty()) {
            int now = st.top(); st.pop();
            pool[now]->cnt += cnt;
            pool[now]->val = 0;
            if(pool[now]->NextNode[0] != -1) pool[now]->val += pool[pool[now]->NextNode[0]]->val;
            if(pool[now]->NextNode[1] != -1) pool[now]->val += pool[pool[now]->NextNode[1]]->val;
        }
    }
    void erase(B bit, C cnt = 1) {
        insert(bit, -cnt);
    }
    C count(B bit) {
        int idx = 0;
        while(pool[idx]->depth != MAXLEN) {
            int depth = pool[idx]->depth;
            B nex = (bit >> (MAXLEN - depth - 1));
            if(pool[idx]->NextNode[nex & 1] == -1) {
                return 0;
            }
            idx = pool[idx]->NextNode[nex & 1];
        }
        return pool[idx]->cnt;
    }
    // 0-indexed
    pair<C, pNode> leftkth_element(C k) {
        assert(0 <= k && k < size());
        int idx = 0;
        while(1) {
            if(pool[idx]->depth == MAXLEN) return make_pair(k, pool[idx]);
            if(pool[idx]->NextNode[0] != -1) {
                if(k < pool[pool[idx]->NextNode[0]]->cnt) {
                    idx = pool[idx]->NextNode[0];
                    continue;
                }
                k -= pool[pool[idx]->NextNode[0]]->cnt;
            }
            if(pool[idx]->NextNode[1] != -1) {
                if(k < pool[pool[idx]->NextNode[1]]->cnt) {
                    idx = pool[idx]->NextNode[1];
                    continue;
                }
                k -= pool[pool[idx]->NextNode[1]]->cnt;
            }
            break;
        }
        return make_pair(k, nullptr);
    }
    pair<C, pNode> rightkth_element(C k) {
        assert(0 <= k);
        auto rec = leftkth_element(pool[0]->cnt - k - 1);
        rec.first = rec.second->cnt - rec.first - 1;
        return rec;
    }
    pNode min_element() {
        return leftkth_element(0).second;
    }
    pNode max_element() {
        return leftkth_element(pool[0]->cnt - 1).second;
    }
    C count_less(B bit) {
        C rec = 0;
        int idx = 0;
        while(pool[idx]->depth != MAXLEN) {
            int depth = pool[idx]->depth;
            B nex = (bit >> (MAXLEN - depth - 1));
            if((nex & 1) == 1 && pool[idx]->NextNode[0] != -1) {
                rec += pool[pool[idx]->NextNode[0]]->cnt;
            }
            if(pool[idx]->NextNode[nex & 1] == -1) break;
            idx = pool[idx]->NextNode[nex & 1];
        }
        return rec;
    }
    C count_nomore(B bit) {
        return count_less(bit + 1);
    }
    C count_more(B bit) {
        return pool[0]->cnt - count_less(bit + 1);
    }
    C count_noless(B bit) {
        return pool[0]->cnt - count_less(bit);
    }
    T query_less(B bit) {
        T rec = 0;
        int idx = 0;
        while(pool[idx]->depth != MAXLEN) {
            int depth = pool[idx]->depth;
            B nex = (bit >> (MAXLEN - depth - 1));
            if((nex & 1) == 1 && pool[idx]->NextNode[0] != -1) {
                rec += pool[pool[idx]->NextNode[0]]->val;
            }
            if(pool[idx]->NextNode[nex & 1] == -1) break;
            idx = pool[idx]->NextNode[nex & 1];
        }
        return rec;
    }
    T query_nomore(B bit) {
        return query_less(bit + 1);
    }
    T query_more(B bit) {
        return pool[0]->val - query_less(bit + 1);
    }
    T query_noless(B bit) {
        return pool[0]->val - query_less(bit);
    }
    // 1-indexed
    T rightkth_query(C c) {
        auto [res, p] = rightkth_element(c - 1);
        return p->key * (res + 1) + query_more(p->key);
    }
    C size() {
        return pool[0]->cnt;
    }
};

/* varify
    diameter : https://atcoder.jp/contests/abc361/submissions/55323033
    LCA      : https://judge.u-aizu.ac.jp/onlinejudge/review.jsp?rid=9459125#1
    Auxiliary: https://atcoder.jp/contests/abc359/submissions/55589339
*/
template<typename C = long long, C MAX = (long long)2e18>
class Tree {
public:
    struct edge {
        int from, to;
        C cost;
    };
    int n;
    bool is_build;
    vector<vector<edge>> G;
    vector<C> dist;
    vector<int> depth;
    vector<vector<int>> parent;
    vector<int> preOrder;
    Tree(int n) : n(n) {
        is_build = false;
        G = vector<vector<edge>>(n);
    };
private:
    void build_dist(int root) {
        dist = vector<C> (n, MAX);
        depth = vector<int> (n, 1e9);
        dist[root] = depth[root] = 0;
        auto dfs = [&](auto self, int now, int par = -1) -> void {
            for(auto [from, to, cost] : G[now]) {
                if(to == par) continue;
                dist[to] = dist[now] + cost;
                depth[to] = depth[now] + 1;
                self(self, to, from);
            }
        };
        dfs(dfs, root);
    }
    void build_lca(int root) {
        int m = 1;
        while((1 << m) < n) m++;
        parent = vector<vector<int>>(m, vector<int>(n, -1));
        auto dfs = [&](auto self, int now, int par = -1) -> void {
            parent[0][now] = par;
            for(auto [from, to, cost] : G[now]) {
                if(to == par) continue;
                self(self, to, from);
            }
        };
        dfs(dfs, root);
        for (int k = 0; k < m - 1; k++) {
            for(int i = 0; i < n; i++) {
                if(parent[k][i] < 0) {
                    parent[k + 1][i] = -1;
                }
                else {
                    parent[k + 1][i] = parent[k][parent[k][i]];
                }
            }
        }
    }
    void build_preOrder(int root) {
        preOrder = vector<int>(n);
        int k = 0;
        auto dfs = [&](auto self, int now, int par = -1) -> void {
            preOrder[now] = k++;
            for(auto [from, to, cost] : G[now]) {
                if(to == par) continue;
                self(self, to, from);
            }
        };
        dfs(dfs, root);
    }
public:
    void add_edge(int a, int b, ll c = 1) {
        G[a].push_back(edge{a, b, c});
        G[b].push_back(edge{b, a, c});
    }
    // O(NlogN)
    void build(int root = 0) {
        build_dist(root);
        build_lca(root);
        build_preOrder(root);
    }
    // O(N)
    pair<C, pair<int, int>> diameter() {
        auto dfs = [&](auto self, int now, int par = -1) -> pair<C, int> {
            pair<C, int> p(0, now);
            for(auto [from, to, cost] : G[now]) {
                if(to == par) continue;
                pair<C, int> rec = self(self, to, from);
                rec.first += cost;
                if(p.first < rec.first) {
                    p = rec;
                }
            }
            return p;
        };
        pair<C, int> mid = dfs(dfs, 0);
        pair<C, int> res = dfs(dfs, mid.second);
        return pair<C, pair<int, int>>(res.first, pair<int, int>(mid.second, res.second));
    }
    // O(logN)
    int lca(int a, int b) {
        if (depth[a] < depth[b]) swap(a, b);
        int m = parent.size();
        for (int k = 0; k < m; k++) {
            if ((depth[a] - depth[b]) >> k & 1) {
                a = parent[k][a];
            }
        }
        if(a == b) return a;
        for (int k = m - 1; k >= 0; k--) {
            if(parent[k][a] != parent[k][b]) {
                a = parent[k][a];
                b = parent[k][b];
            }
        }
        return parent[0][a];
    }
    // O(logN)
    C get_dist(int a, int b) {
        return dist[a] + dist[b] - dist[lca(a, b)] * 2;
    }
    // O(logN)
    C is_on_path(int l, int r, int mid) {
        return get_dist(l, mid) + get_dist(mid, r) == get_dist(l, r);
    }
    // O(|vs|logN)
    int auxiliary(vector<int>& vs, unordered_map<int, vector<int>>& rec) {
        int m = vs.size();
        sort(vs.begin(), vs.end(), [this](int a, int b){return preOrder[a] < preOrder[b];});
        stack<int> st;
        st.push(vs[0]);
        for(int i = 1; i < m; i++) {
            int w = this->lca(vs[i - 1], vs[i]);
            if(w != vs[i - 1]) {
                int last = st.top(); st.pop();
                while(st.size() && depth[w] < depth[st.top()]) {
                    rec[st.top()].push_back(last);
                    last = st.top(); st.pop();
                }
                if(st.empty() || st.top() != w) {
                    st.push(w);
                    vs.push_back(w);
                    rec[w].push_back(last);
                }
                else {
                    rec[w].push_back(last);
                }
            }
            st.push(vs[i]);
        }
        if(st.size() > 1) {
            int last = st.top(); st.pop();
            while(st.size() > 1) {
                rec[st.top()].push_back(last);
                last = st.top();
                st.pop();
            }
            rec[st.top()].push_back(last);
        }
        return st.top();
    }
};
